# 文档与代码同步方案

## 📋 问题分析

### 当前存在的问题

1. **文档滞后于代码**：技术文档中的信息与实际代码实现不一致
2. **手动维护成本高**：需要开发者手动更新文档，容易遗漏
3. **版本不同步**：代码版本更新后，文档没有及时跟进
4. **信息分散**：技术信息散落在代码、注释、文档等多个地方

### 根本原因

- **缺乏自动化机制**：没有建立代码变更时自动更新文档的流程
- **文档责任不明确**：开发流程中缺少文档维护的明确责任分工
- **检查机制缺失**：没有在CI/CD中集成文档一致性检查

## 🎯 解决方案

### 方案一：自动化文档生成（推荐）

#### 1.1 技术选型

- **TypeDoc**：从 TypeScript 代码自动生成 API 文档
- **JSDoc**：代码注释标准化
- **Markdown 模板引擎**：动态生成文档内容
- **GitHub Actions**：自动化执行文档更新

**步骤2：创建文档生成脚本**

```typescript
// scripts/generate-docs.ts
import { exec } from 'child_process';
import { readFileSync, writeFileSync } from 'fs';
import { promisify } from 'util';

const execAsync = promisify(exec);

interface ProjectInfo {
  version: string;
  dependencies: Record<string, string>;
  devDependencies: Record<string, string>;
}

async function generateTechDoc() {
  console.log('🔄 开始生成技术文档...');

  // 1. 读取项目信息
  const packageJson: ProjectInfo = JSON.parse(
    readFileSync('package.json', 'utf-8')
  );

  // 2. 生成 API 文档
  await execAsync('npx typedoc');

  // 3. 提取组件信息
  const componentInfo = await extractComponentInfo();

  // 4. 提取状态管理信息
  const storeInfo = await extractStoreInfo();

  // 5. 生成技术栈信息
  const techStack = generateTechStackInfo(packageJson);

  // 6. 更新技术文档模板
  await updateTechDocTemplate({
    version: packageJson.version,
    techStack,
    componentInfo,
    storeInfo,
    lastUpdated: new Date().toISOString().split('T')[0],
  });

  console.log('✅ 技术文档生成完成');
}

async function extractComponentInfo() {
  // 扫描组件目录，提取组件信息
  const { stdout } = await execAsync(
    'find src/components -name "*.tsx" -type f'
  );

  const componentFiles = stdout.trim().split('\n');
  const components = [];

  for (const file of componentFiles) {
    if (file.includes('__tests__')) continue;

    const content = readFileSync(file, 'utf-8');
    const componentName = file.split('/').pop()?.replace('.tsx', '');

    // 提取组件导出信息
    const hasDefaultExport = /export default/.test(content);
    const namedExports = content.match(/export \{([^}]+)\}/g) || [];

    components.push({
      name: componentName,
      path: file,
      hasDefaultExport,
      namedExports: namedExports.length,
      isLazy: content.includes('React.lazy'),
    });
  }

  return components;
}

async function extractStoreInfo() {
  // 扫描 store 目录，提取状态管理信息
  const { stdout } = await execAsync('find src/store -name "*.ts" -type f');
  const storeFiles = stdout.trim().split('\n');
  const stores = [];

  for (const file of storeFiles) {
    if (file.includes('__tests__')) continue;

    const content = readFileSync(file, 'utf-8');
    const storeName = file.split('/').pop()?.replace('.ts', '');

    // 提取 Zustand store 信息
    const hasCreate = /create</.test(content);
    const stateProps = content.match(/^\s*(\w+):\s*[^,]+,?$/gm) || [];
    const methods = content.match(/^\s*(\w+):\s*\([^)]*\)\s*=>/gm) || [];

    stores.push({
      name: storeName,
      path: file,
      isZustandStore: hasCreate,
      stateProperties: stateProps.length,
      methods: methods.length,
    });
  }

  return stores;
}

function generateTechStackInfo(packageJson: ProjectInfo) {
  const dependencies = packageJson.dependencies || {};
  const devDependencies = packageJson.devDependencies || {};

  return {
    react: dependencies.react || 'Unknown',
    typescript: devDependencies.typescript || 'Unknown',
    vite: devDependencies.vite || 'Unknown',
    antd: dependencies.antd || 'Unknown',
    zustand: dependencies.zustand || 'Unknown',
    vitest: devDependencies.vitest || 'Unknown',
    eslint: devDependencies.eslint || 'Unknown',
    prettier: devDependencies.prettier || 'Unknown',
  };
}

async function updateTechDocTemplate(data: any) {
  const template = readFileSync('docs/templates/技术文档.template.md', 'utf-8');

  let updatedDoc = template
    .replace(/{{version}}/g, data.version)
    .replace(/{{lastUpdated}}/g, data.lastUpdated)
    .replace(/{{reactVersion}}/g, data.techStack.react)
    .replace(/{{typescriptVersion}}/g, data.techStack.typescript)
    .replace(/{{viteVersion}}/g, data.techStack.vite)
    .replace(/{{antdVersion}}/g, data.techStack.antd)
    .replace(/{{zustandVersion}}/g, data.techStack.zustand)
    .replace(/{{vitestVersion}}/g, data.techStack.vitest);

  // 生成组件列表
  const componentList = data.componentInfo
    .map((comp: any) => `- **${comp.name}**${comp.isLazy ? ' (懒加载)' : ''}`)
    .join('\n');
  updatedDoc = updatedDoc.replace(/{{componentList}}/g, componentList);

  // 生成状态管理列表
  const storeList = data.storeInfo
    .map(
      (store: any) =>
        `- **${store.name}**：${store.stateProperties} 个状态属性，${store.methods} 个方法`
    )
    .join('\n');
  updatedDoc = updatedDoc.replace(/{{storeList}}/g, storeList);

  writeFileSync('docs/技术文档.md', updatedDoc);
}

// 执行文档生成
generateTechDoc().catch(console.error);
```

on:
  push:
    branches: [main, develop]
    paths:
      - 'src/**'
      - 'package.json'
      - 'docs/templates/**'
  pull_request:
    branches: [main]
    paths:
      - 'src/**'
      - 'package.json'

jobs:
  docs-sync:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v4
        with:
          token: ${{ secrets.GITHUB_TOKEN }}

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '18'
          cache: 'pnpm'

      - name: Install dependencies
        run: pnpm install

      - name: Generate documentation
        run: pnpm docs:generate

      - name: Check for documentation changes
        id: docs-changed
        run: |
          if git diff --quiet docs/技术文档.md; then
            echo "changed=false" >> $GITHUB_OUTPUT
          else
            echo "changed=true" >> $GITHUB_OUTPUT
          fi

      - name: Commit documentation updates
        if: steps.docs-changed.outputs.changed == 'true'
        run: |
          git config --local user.email "action@github.com"
          git config --local user.name "GitHub Action"
          git add docs/技术文档.md docs/api/
          git commit -m "📝 自动更新技术文档"
          git push

### 方案二：文档一致性检查

#### 2.1 创建文档验证脚本

```typescript
// scripts/validate-docs.ts
import { readFileSync } from 'fs';
import { glob } from 'glob';

interface ValidationResult {
  isValid: boolean;
  errors: string[];
  warnings: string[];
}

async function validateDocumentation(): Promise<ValidationResult> {
  const result: ValidationResult = {
    isValid: true,
    errors: [],
    warnings: [],
  };

  // 1. 检查技术栈版本一致性
  await validateTechStackVersions(result);

  // 2. 检查组件文档完整性
  await validateComponentDocs(result);

  // 3. 检查 API 文档一致性
  await validateApiDocs(result);

  return result;
}

async function validateTechStackVersions(result: ValidationResult) {
  const packageJson = JSON.parse(readFileSync('package.json', 'utf-8'));
  const techDoc = readFileSync('docs/技术文档.md', 'utf-8');

  // 检查版本号一致性
  const dependencies = {
    ...packageJson.dependencies,
    ...packageJson.devDependencies,
  };

  const versionChecks = [
    { name: 'React', key: 'react', pattern: /React (\d+\.\d+\.\d+)/ },
    {
      name: 'TypeScript',
      key: 'typescript',
      pattern: /TypeScript (\d+\.\d+\.\d+)/,
    },
    { name: 'Vite', key: 'vite', pattern: /Vite (\d+\.\d+\.\d+)/ },
  ];

  for (const check of versionChecks) {
    const actualVersion = dependencies[check.key];
    const docMatch = techDoc.match(check.pattern);

    if (!docMatch) {
      result.errors.push(`文档中缺少 ${check.name} 版本信息`);
      result.isValid = false;
    } else if (docMatch[1] !== actualVersion?.replace('^', '')) {
      result.errors.push(
        `${check.name} 版本不一致：文档显示 ${docMatch[1]}，实际为 ${actualVersion}`
      );
      result.isValid = false;
    }
  }
}

async function validateComponentDocs(result: ValidationResult) {
  // 获取所有组件文件
  const componentFiles = await glob('src/components/**/*.tsx', {
    ignore: ['**/__tests__/**', '**/*.test.tsx', '**/*.spec.tsx'],
  });

  const techDoc = readFileSync('docs/技术文档.md', 'utf-8');

  for (const file of componentFiles) {
    const componentName = file.split('/').pop()?.replace('.tsx', '');
    if (!componentName) continue;

    // 检查组件是否在文档中提及
    if (!techDoc.includes(componentName)) {
      result.warnings.push(`组件 ${componentName} 未在技术文档中提及`);
    }
  }
}

async function validateApiDocs(result: ValidationResult) {
  // 检查 store 接口是否有对应文档
  const storeFiles = await glob('src/store/*.ts', {
    ignore: ['**/__tests__/**', '**/*.test.ts'],
  });

  const techDoc = readFileSync('docs/技术文档.md', 'utf-8');

  for (const file of storeFiles) {
    const storeName = file.split('/').pop()?.replace('.ts', '');
    if (!storeName) continue;

    if (!techDoc.includes(storeName)) {
      result.warnings.push(`状态管理 ${storeName} 未在技术文档中描述`);
    }
  }
}

// 执行验证
validateDocumentation().then(result => {
  console.log('\n📋 文档验证结果:');

  if (result.errors.length > 0) {
    console.log('\n❌ 错误:');
    result.errors.forEach(error => console.log(`  - ${error}`));
  }

  if (result.warnings.length > 0) {
    console.log('\n⚠️  警告:');
    result.warnings.forEach(warning => console.log(`  - ${warning}`));
  }

  if (result.isValid && result.warnings.length === 0) {
    console.log('\n✅ 文档验证通过');
  }

  process.exit(result.isValid ? 0 : 1);
});
```
